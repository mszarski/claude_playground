set(_dsp_sources src/fft.cpp src/baseband.cpp src/conditioning.cpp src/pulse_compression.cpp)

  list(APPEND _dsp_sources src/dsp_cuda.cu)
endif()

  target_link_libraries(sonar_dsp PRIVATE CUDA::cufft)
endif()
