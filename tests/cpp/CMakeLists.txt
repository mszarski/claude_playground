include(GoogleTest)

function(sonar_add_test name)
  add_executable(${name} ${ARGN})
  target_link_libraries(${name} PRIVATE sonar::core sonar::io sonar::sim GTest::gtest GTest::gtest_main)
  gtest_discover_tests(${name} DISCOVERY_TIMEOUT 60)
endfunction()
